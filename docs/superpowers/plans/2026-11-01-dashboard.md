- [ ] **Step 2: Add the complete fixture and minimal calculations.**

```ts
export type Tank = {
  id: string
  name: string
  plant: string
  phase: string
  status: 'normal' | 'warning'
  water: number
  ppm: number
  temperature: number
  battery: number
}

export const tanks: Tank[] = [
  { id: 'a', name: 'Tandon A', plant: 'Selada', phase: 'Vegetatif', status: 'normal', water: 78, ppm: 1250, temperature: 25.5, battery: 90 },
  { id: 'b', name: 'Tandon B', plant: 'Pakcoy', phase: 'Vegetatif', status: 'normal', water: 65, ppm: 850, temperature: 27.2, battery: 72 },
  { id: 'c', name: 'Tandon C', plant: 'Kangkung', phase: 'Panen', status: 'warning', water: 32, ppm: 980, temperature: 26.1, battery: 38 },
  { id: 'd', name: 'Tandon D', plant: 'Selada', phase: 'Pembibitan', status: 'normal', water: 84, ppm: 640, temperature: 24.8, battery: 96 },
]

export function getTankSummary(tanks: readonly Tank[]): {
  total: number
  normal: number
  warning: number
} {
  return {
    total: tanks.length,
    normal: tanks.filter((tank) => tank.status === 'normal').length,
    warning: tanks.filter((tank) => tank.status === 'warning').length,
  }
}

export const numberFormatter = new Intl.NumberFormat('id-ID')
export const temperatureFormatter = new Intl.NumberFormat('id-ID', {
  minimumFractionDigits: 1,
  maximumFractionDigits: 1,
})
```

```tsx
import { Battery, Droplets, Gauge, Thermometer } from 'lucide-react'
import { Badge } from '#/components/ui/badge'
import { Card, CardAction, CardContent, CardDescription, CardHeader, CardTitle } from '#/components/ui/card'
import { type Tank, numberFormatter, temperatureFormatter } from '#/data/tanks'

export function TankCard({ tank }: { tank: Tank }) {
  const metrics = [
    { label: 'Air', value: `${numberFormatter.format(tank.water)}%`, icon: Droplets },
    { label: 'PPM', value: numberFormatter.format(tank.ppm), icon: Gauge },
    { label: 'Suhu', value: `${temperatureFormatter.format(tank.temperature)}°C`, icon: Thermometer },
    { label: 'Baterai', value: `${numberFormatter.format(tank.battery)}%`, icon: Battery },
  ]

  return (
    <Card>
      <CardHeader>
        <CardTitle><h3>{tank.name}</h3></CardTitle>
        <CardDescription>{tank.plant} · Fase {tank.phase.toLowerCase()}</CardDescription>
        <CardAction>
          <Badge variant={tank.status === 'warning' ? 'destructive' : 'secondary'}>
            {tank.status === 'warning' ? 'Warning' : 'Normal'}
          </Badge>
        </CardAction>
      </CardHeader>
      <CardContent>
        <dl className="grid grid-cols-2 gap-x-6 gap-y-5">
          {metrics.map(({ label, value, icon: Icon }) => (
            <div key={label}>
              <dt className="flex items-center gap-2 text-sm text-muted-foreground">
                <Icon className="size-4" aria-hidden="true" />
                {label}
              </dt>
              <dd className="mt-1 text-xl font-semibold tabular-nums">{value}</dd>
            </div>
          ))}
        </dl>
      </CardContent>
    </Card>
  )
}
```

- [ ] **Step 2: Replace the marketing starter route with the full dashboard.**

```tsx
import { createFileRoute } from '@tanstack/react-router'
import { TankCard } from '#/components/tank-card'
import { Badge } from '#/components/ui/badge'
import { Card, CardContent } from '#/components/ui/card'
import { Empty, EmptyDescription, EmptyHeader, EmptyTitle } from '#/components/ui/empty'
import { getTankSummary, tanks } from '#/data/tanks'

export const Route = createFileRoute('/')({ component: Dashboard })

function Dashboard() {
  const summary = getTankSummary(tanks)
  const stats = [
    { label: 'Tandon', value: summary.total },
    { label: 'Normal', value: summary.normal },
    { label: 'Warning', value: summary.warning },
  ]

  return (
    <main className="mx-auto flex w-full max-w-3xl flex-col gap-6 px-4 py-6">
      <header className="flex items-start justify-between gap-4">
        <div className="min-w-0">
          <h1 className="text-2xl font-semibold tracking-tight">HydroMonitor</h1>
          <p className="mt-1 text-sm text-muted-foreground">Monitoring seluruh tandon</p>
        </div>
        <Badge variant="outline">Data demo</Badge>
      </header>

      <section aria-label="Ringkasan kondisi tandon">
        <Card>
          <CardContent>
            <dl className="grid grid-cols-3 divide-x divide-border">
              {stats.map(({ label, value }) => (
                <div key={label} className="flex flex-col items-center gap-1">
                  <dt className="text-xs text-muted-foreground">{label}</dt>
                  <dd className="order-first text-2xl font-semibold tabular-nums">{value}</dd>
                </div>
              ))}
            </dl>
          </CardContent>
        </Card>
      </section>

      <section aria-labelledby="tanks-heading" className="flex flex-col gap-4">
        <h2 id="tanks-heading" className="text-lg font-semibold">Daftar Tandon</h2>
        {tanks.length === 0 ? (
          <Empty>
            <EmptyHeader>
              <EmptyTitle>Belum ada tandon</EmptyTitle>
              <EmptyDescription>Belum ada data tandon untuk ditampilkan.</EmptyDescription>
            </EmptyHeader>
          </Empty>
        ) : (
          <div className="flex flex-col gap-4">
            {tanks.map((tank) => <TankCard key={tank.id} tank={tank} />)}
          </div>
        )}
      </section>
    </main>
  )
}
```
